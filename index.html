01
Arquitectura Dual INA226
Solución para rango 3V–100V · Selección automática de canal por firmware
Problema resuelto: El INA226 tiene límite de 36V en el bus. Un panel de 48V nominal genera Voc de hasta 60V en frío (β=−0.30%/°C). La arquitectura dual usa un divisor resistivo 820kΩ/180kΩ en el Canal A para escalar 100V → 18V, y el Canal B conecta directamente para mini paneles con alta resolución.
CANAL A — ALTO VOLTAJE (0–100V / 0–8A) Panel [+] ──[Fusible 10A DC]──[TVS P6KE100CA]──┬──[R1=820kΩ]──[INA226_A V+] │ I2C Addr: 0x40 [Shunt 10mΩ]──[INA226_A Shunt] │ Panel [-] ───────────────────────────────────────┴──[R2=180kΩ]──GND V_real = V_chip × 5.556 | Factor K_div = (820+180)/180 = 5.556× Rango: 0–100V panel → 0–18V en chip ✓ Límite chip: 36V ✓ CANAL B — BAJO VOLTAJE (0–36V / 0–0.82A) Mini [+] ──[Fusible 1A DC]──[Shunt 100mΩ]──[INA226_B V+] ← Sin divisor └──────[INA226_B Shunt] I2C Addr: 0x41 Mini [-] ────────────────────────────────────────────────────GND SELECCIÓN AUTOMÁTICA DE CANAL: V_raw × 5.556 > 10V → Canal A | V_raw × 5.556 ≤ 10V → Canal B ESP32 — I2C compartido (SDA GPIO21 / SCL GPIO22): ├── INA226_A [0x40] — Canal alto voltaje ├── INA226_B [0x41] — Canal bajo voltaje ├── MCP4725 [0x62] — DAC control MOSFET barrido I-V └── SSD1306 [0x3C] — Pantalla OLED 128×64 1-Wire GPIO4: DS18B20 temperatura ambiente ADC GPIO34: NTC 10kΩ temperatura célula DUT ADC GPIO35: Célula referencia irradiancia (mini panel 5V) GPIO26: Relé Canal A GPIO27: Relé Canal B GPIO25: Gate MOSFET (vía opamp TL071 → 0–10V)

// ═══════════════════════════════════════════════════════════════════ // BANCO SOLAR FV — Firmware ESP32 v2.0 // Arquitectura Dual INA226: Canal A (0–100V) + Canal B (0–36V) // Pantalla OLED SSD1306 128×64 + WiFi Dashboard // San Juan, Argentina · IEC 60904-1 · IEC 62446-1 // Ing. Germán Golder — Factor Solar® © 2025 // ═══════════════════════════════════════════════════════════════════ #include <WiFi.h> #include <WebServer.h> #include <Wire.h> #include <INA226.h> #include <Adafruit_SSD1306.h> #include <Adafruit_GFX.h> #include <Adafruit_MCP4725.h> #include <OneWire.h> #include <DallasTemperature.h> #include <ArduinoJson.h> // ── WiFi ────────────────────────────────────────────────────────── const char* SSID = "TU_WIFI"; const char* PASSWORD = "TU_CLAVE"; // ── Pines ───────────────────────────────────────────────────────── #define PIN_DS18B20 4 // Temperatura ambiente 1-Wire #define PIN_NTC 34 // Temperatura célula DUT (ADC) #define PIN_CELREF 35 // Célula referencia irradiancia (ADC) #define PIN_RELE_A 26 // Relé Canal A #define PIN_RELE_B 27 // Relé Canal B #define PIN_GATE 25 // Gate MOSFET (vía opamp) #define OLED_W 128 #define OLED_H 64 // ── Calibración Canal A (alto voltaje) ──────────────────────────── const float K_DIV = 5.556f; // Factor divisor R1/R2=(820k+180k)/180k const float R_SHUNT_A = 0.010f; // 10mΩ → I_max 8.2A (usar 0.005 para 16A) const float I_MAX_A = 8.2f; // ── Calibración Canal B (bajo voltaje) ─────────────────────────── const float R_SHUNT_B = 0.100f; // 100mΩ → I_max 0.82A const float I_MAX_B = 0.82f; // ── Coeficientes de temperatura Si cristalino ──────────────────── const float ALPHA = 0.0005f; // +0.05%/°C Isc const float BETA = -0.003f; // -0.30%/°C Voc const float GAMMA = -0.0043f; // -0.43%/°C Pmpp // ── Factor irradiancia célula ref. ─────────────────────────────── float FACTOR_K_CEL = 11.76f; // W/m² por mA — calibrar vs. sol real // ── N puntos barrido ───────────────────────────────────────────── const int N_PTS = 100; // ── Objetos ─────────────────────────────────────────────────────── INA226 inaA(0x40); INA226 inaB(0x41); Adafruit_SSD1306 oled(OLED_W, OLED_H, &Wire, -1); Adafruit_MCP4725 dac; OneWire ow(PIN_DS18B20); DallasTemperature tempSensor(&ow); WebServer server(80); // ── Variables globales ─────────────────────────────────────────── enum Canal { CANAL_A, CANAL_B }; Canal canalActivo = CANAL_A; float iv_v[N_PTS], iv_i[N_PTS]; float voc, isc, pmpp, vmpp, impp, ff; float voc_stc, isc_stc, pmpp_stc, vmpp_stc, impp_stc; float g_irrad, t_celda, t_amb; bool barridoListo = false; int progreso = 0; bool barridoActivo= false; // ══════════════════════════════════════════════════════════════════ // SETUP // ══════════════════════════════════════════════════════════════════ void setup() { Serial.begin(115200); Wire.begin(21, 22); // SDA=21, SCL=22 // ── OLED ────────────────────────────────────────────────────── if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { Serial.println("OLED no encontrado"); } oled.clearDisplay(); oled.setTextColor(SSD1306_WHITE); oled.setTextSize(1); oled.setCursor(0,0); oled.println("BANCO SOLAR FV v2.0"); oled.println("Ing. German Golder"); oled.println("Iniciando..."); oled.display(); // ── INA226 Canal A ──────────────────────────────────────────── inaA.begin(); inaA.configure(INA226_AVERAGES_16, INA226_BUS_CONV_TIME_1100US, INA226_SHUNT_CONV_TIME_1100US, INA226_MODE_SHUNT_BUS_CONT); inaA.calibrate(R_SHUNT_A, I_MAX_A); // ── INA226 Canal B ──────────────────────────────────────────── inaB.begin(); inaB.configure(INA226_AVERAGES_16, INA226_BUS_CONV_TIME_1100US, INA226_SHUNT_CONV_TIME_1100US, INA226_MODE_SHUNT_BUS_CONT); inaB.calibrate(R_SHUNT_B, I_MAX_B); // ── DAC ─────────────────────────────────────────────────────── dac.begin(0x62); dac.setVoltage(0, false); // ── Temperatura ─────────────────────────────────────────────── tempSensor.begin(); // ── Pines ───────────────────────────────────────────────────── pinMode(PIN_RELE_A, OUTPUT); digitalWrite(PIN_RELE_A, LOW); pinMode(PIN_RELE_B, OUTPUT); digitalWrite(PIN_RELE_B, LOW); pinMode(PIN_GATE, OUTPUT); analogWrite(PIN_GATE, 0); // ── WiFi ────────────────────────────────────────────────────── WiFi.begin(SSID, PASSWORD); oled.clearDisplay(); oled.setCursor(0,0); oled.println("Conectando WiFi..."); oled.display(); while (WiFi.status() != WL_CONNECTED) { delay(300); } Serial.println("IP: " + WiFi.localIP().toString()); // ── Mostrar IP en OLED ──────────────────────────────────────── oled.clearDisplay(); oled.setCursor(0,0); oled.println("WiFi OK!"); oled.println(WiFi.localIP().toString()); oled.println("Dashboard disponible"); oled.display(); delay(2000); // ── Rutas servidor web ──────────────────────────────────────── server.on("/", handleRoot); server.on("/estado", handleEstado); server.on("/barrido", handleBarrido); server.on("/curva", handleCurva); server.on("/csv", handleCSV); server.on("/progreso", handleProgreso); server.begin(); } // ══════════════════════════════════════════════════════════════════ // LOOP // ══════════════════════════════════════════════════════════════════ void loop() { server.handleClient(); leerSensores(); actualizarOLED(); delay(500); } // ══════════════════════════════════════════════════════════════════ // SELECCIÓN AUTOMÁTICA DE CANAL // ══════════════════════════════════════════════════════════════════ void seleccionarCanal() { float v_raw = inaA.readBusVoltage(); float v_real = v_raw * K_DIV; if (v_real > 10.0) { canalActivo = CANAL_A; Serial.printf("Canal A: %.1fV\n", v_real); } else { canalActivo = CANAL_B; Serial.printf("Canal B: %.2fV\n", inaB.readBusVoltage()); } } // ══════════════════════════════════════════════════════════════════ // LECTURA DE SENSORES // ══════════════════════════════════════════════════════════════════ void leerSensores() { // Irradiancia desde célula de referencia int raw_cel = analogRead(PIN_CELREF); float v_cel = raw_cel * 3.3f / 4095.0f; // 0–3.3V ESP32 float i_cel_mA = v_cel / 0.100f * 1000.0f; // shunt 100mΩ g_irrad = i_cel_mA * FACTOR_K_CEL; // Temperatura ambiente DS18B20 tempSensor.requestTemperatures(); t_amb = tempSensor.getTempCByIndex(0); // Temperatura célula NTC (Steinhart-Hart simplificado) int raw_ntc = analogRead(PIN_NTC); float r_ntc = 10000.0f * raw_ntc / (4095.0f - raw_ntc); t_celda = 1.0f / (log(r_ntc / 10000.0f) / 3950.0f + 1.0f/298.15f) - 273.15f; } // ══════════════════════════════════════════════════════════════════ // LECTURA V/I SEGÚN CANAL ACTIVO // ══════════════════════════════════════════════════════════════════ float leerVoltaje() { if (canalActivo == CANAL_A) return inaA.readBusVoltage() * K_DIV; // Aplica factor divisor else return inaB.readBusVoltage(); // Directo sin divisor } float leerCorriente() { if (canalActivo == CANAL_A) return inaA.readShuntCurrent(); else return inaB.readShuntCurrent(); } // ══════════════════════════════════════════════════════════════════ // BARRIDO I-V // ══════════════════════════════════════════════════════════════════ void ejecutarBarrido() { barridoActivo = true; barridoListo = false; progreso = 0; seleccionarCanal(); // Activar relé del canal correspondiente digitalWrite(canalActivo == CANAL_A ? PIN_RELE_A : PIN_RELE_B, HIGH); delay(200); // Medir Voc (carga al mínimo) dac.setVoltage(0, false); delay(300); voc = leerVoltaje(); // Medir Isc (carga al máximo) dac.setVoltage(4095, false); delay(100); isc = leerCorriente(); // Barrido de N_PTS puntos float p_max = 0; for (int k = 0; k < N_PTS; k++) { uint16_t dac_val = (uint16_t)(k * 4095.0f / (N_PTS - 1)); dac.setVoltage(dac_val, false); delay(80); iv_v[k] = leerVoltaje(); iv_i[k] = leerCorriente(); float p = iv_v[k] * iv_i[k]; if (p > p_max) { p_max = p; pmpp = p; vmpp = iv_v[k]; impp = iv_i[k]; } progreso = (k + 1) * 100 / N_PTS; oledBarrido(k + 1, iv_v[k], iv_i[k], p); // Actualizar OLED server.handleClient(); // Atender WiFi durante barrido } // Factor de Forma ff = pmpp / (voc * isc); // Corrección a STC — IEC 60904-1 float dT = 25.0f - t_celda; float Ki = g_irrad > 10 ? 1000.0f / g_irrad : 1.0f; voc_stc = voc * (1.0f + BETA * dT); isc_stc = isc * Ki * (1.0f + ALPHA * dT); vmpp_stc = vmpp * (1.0f + BETA * dT); impp_stc = impp * Ki * (1.0f + ALPHA * dT); pmpp_stc = pmpp * Ki * (1.0f + GAMMA * dT); dac.setVoltage(0, false); digitalWrite(PIN_RELE_A, LOW); digitalWrite(PIN_RELE_B, LOW); barridoActivo = false; barridoListo = true; oledResultado(); // Mostrar resultado en OLED } // ══════════════════════════════════════════════════════════════════ // PANTALLA OLED — funciones // ══════════════════════════════════════════════════════════════════ void actualizarOLED() { if (barridoActivo) return; // Durante barrido lo maneja oledBarrido() oled.clearDisplay(); oled.setTextSize(1); oled.setCursor(0, 0); oled.print("BANCO FV"); oled.setCursor(80, 0); oled.print(WiFi.status() == WL_CONNECTED ? "* WiFi" : " --"); // Línea separadora oled.drawLine(0, 9, 127, 9, SSD1306_WHITE); // Irradiancia y temperatura oled.setCursor(0, 12); oled.printf("G=%4.0fW/m2 T=%4.1fC", g_irrad, t_celda); // Voc / Isc oled.setCursor(0, 22); if (barridoListo) { oled.printf("Vo=%5.2fV I=%5.3fA", voc_stc, isc_stc); oled.setCursor(0, 32); oled.printf("Pp=%5.1fW FF=%.3f", pmpp_stc, ff); oled.setCursor(0, 42); oled.printf("G=%.1fW K=%.3f", g_irrad, g_irrad>10 ? 1000.0f/g_irrad : 0); oled.setCursor(0, 54); oled.print(ff >= 0.65 ? "APTO *" : "REVISAR !"); oled.setCursor(72, 54); oled.print(canalActivo == CANAL_A ? "[CAN A]" : "[CAN B]"); } else { oled.print("En espera..."); oled.setCursor(0, 32); oled.print("Presione Barrido"); oled.setCursor(0, 42); oled.printf("IP: %s", WiFi.localIP().toString().c_str()); } oled.display(); } void oledBarrido(int punto, float v, float i, float p) { oled.clearDisplay(); oled.setCursor(0, 0); oled.print("BARRIDO I-V"); oled.setCursor(78, 0); oled.printf("%d%%", progreso); // Barra de progreso oled.drawRect(0, 10, 128, 8, SSD1306_WHITE); oled.fillRect(1, 11, progreso * 126 / 100, 6, SSD1306_WHITE); oled.setCursor(0, 22); oled.printf("V=%5.2fV I=%6.3fA", v, i); oled.setCursor(0, 32); oled.printf("P = %6.2f W", p); oled.setCursor(0, 44); oled.printf("Punto %d/%d", punto, N_PTS); oled.setCursor(72, 44); oled.print(canalActivo == CANAL_A ? "CAN A" : "CAN B"); oled.display(); } void oledResultado() { oled.clearDisplay(); oled.setCursor(0, 0); oled.print("RESULTADO STC"); oled.drawLine(0, 9, 127, 9, SSD1306_WHITE); oled.setCursor(0, 12); oled.printf("Pmpp = %6.1f W", pmpp_stc); oled.setCursor(0, 22); oled.printf("FF = %.3f", ff); oled.setCursor(0, 32); oled.printf("Vo=%.1fV Is=%.3fA", voc_stc, isc_stc); oled.setCursor(0, 44); oled.print(ff >= 0.65 ? ">>> PANEL APTO <<<" : ">>> REVISAR !!!"); oled.display(); } // ══════════════════════════════════════════════════════════════════ // HANDLERS WEB // ══════════════════════════════════════════════════════════════════ void handleEstado() { StaticJsonDocument<512> doc; doc["voc"] = leerVoltaje(); doc["isc"] = leerCorriente(); doc["g"] = g_irrad; doc["t_celda"] = t_celda; doc["t_amb"] = t_amb; doc["canal"] = canalActivo == CANAL_A ? "A" : "B"; doc["listo"] = barridoListo; doc["activo"] = barridoActivo; doc["progreso"]= progreso; String json; serializeJson(doc, json); server.sendHeader("Access-Control-Allow-Origin", "*"); server.send(200, "application/json", json); } void handleBarrido() { server.send(200, "text/plain", "INICIANDO"); ejecutarBarrido(); } void handleCurva() { String json = "{\"v\":["; for (int k=0; k<N_PTS; k++) { if(k) json+=","; json+=String(iv_v[k],3); } json += "],\"i\":["; for (int k=0; k<N_PTS; k++) { if(k) json+=","; json+=String(iv_i[k],3); } json += "],\"voc\":" + String(voc_stc,2); json += ",\"isc\":" + String(isc_stc,3); json += ",\"pmpp\":" + String(pmpp_stc,1); json += ",\"vmpp\":" + String(vmpp_stc,2); json += ",\"impp\":" + String(impp_stc,3); json += ",\"ff\":" + String(ff,3); json += ",\"g\":" + String(g_irrad,0); json += ",\"tc\":" + String(t_celda,1); json += ",\"canal\":"+ (canalActivo==CANAL_A ? "\"A\"" : "\"B\"") + "}"; server.send(200, "application/json", json); } void handleProgreso() { String json = "{\"p\":" + String(progreso) + ",\"a\":"; json += barridoActivo ? "true}" : "false}"; server.send(200, "application/json", json); } void handleCSV() { String csv = "V(V),I(A),P(W)\r\n"; for(int k=0; k<N_PTS; k++) csv += String(iv_v[k],3)+","+String(iv_i[k],3)+","+String(iv_v[k]*iv_i[k],2)+"\r\n"; server.sendHeader("Content-Disposition", "attachment;filename=curva_iv.csv"); server.send(200, "text/csv", csv); } void handleRoot() { server.send(200, "text/html", "<!-- Dashboard HTML desde SPIFFS -->"); }
